function scrollToGallery() {
  document.getElementById("gallery").scrollIntoView({ behavior: "smooth" });
}

fetch("videos.json")
  .then(res => res.json())
  .then(data => {
    const carousel = document.getElementById("videoCarousel");
    data.videos.forEach(video => {
      const iframe = document.createElement("iframe");
      iframe.src = `https://www.youtube.com/embed/${video.id}`;
      iframe.allowFullscreen = true;
      carousel.appendChild(iframe);
    });
  });

⚙️ script.js (tambahan fungsi load Playlist)

javascript
// Load video gallery
function scrollToGallery() {
  document.getElementById("gallery").scrollIntoView({ behavior: "smooth" });
}

fetch("videos.json")
  .then(res => res.json())
  .then(data => {
    const carousel = document.getElementById("videoCarousel");
    data.videos.forEach(video => {
      const iframe = document.createElement("iframe");
      iframe.src = `https://www.youtube.com/embed/${video.id}`;
      iframe.allowFullscreen = true;
      carousel.appendChild(iframe);
    });
  });

// Load playlist komunitas
fetch("playlist.json")
  .then(res => res.json())
  .then(data => {
    const container = document.getElementById("playlistContainer");
    data.playlist.forEach(track => {
      const li = document.createElement("li");
      li.innerHTML = `<span>${track.title} - ${track.artist}</span>
                      <a href="${track.url}" target="_blank">Play</a>`;
      container.appendChild(li);
    });
  });

Untuk bisa download website Studio yang sudah kita buat, kamu tinggal ikuti langkah‑langkah berikut supaya semua file (HTML, CSS, JS, JSON) bisa dikompresi jadi ZIP dan siap di‑upload ke GitHub Pages.

01

Buat Folder Project

Kumpulkan semua file website ke dalam satu folder.

  • Buat folder bernama studio

  • Simpan file index.html, style.css, script.js, videos.json, dan playlist.json di dalamnya

02

Compress ke ZIP

File ZIP jadi

Ubah folder project menjadi file ZIP agar mudah dibagikan atau di‑upload.

  • Windows: klik kanan folder → Send to → Compressed (zipped) folder

  • Mac: klik kanan folder → Compress 'studio'

  • Linux: gunakan perintah zip -r studio.zip studio/ di terminal

03

Upload ke GitHub

Gunakan file ZIP untuk deploy ke GitHub Pages.

  • Masuk ke akun GitHub

  • Buat repository baru bernama studio

  • Upload file ZIP lalu Extract di repo

  • Aktifkan GitHub Pages di Settings → Pages → pilih branch main dan folder /root

04

Akses Website

Setelah GitHub Pages aktif, website bisa diakses publik.

  • Tunggu beberapa menit

  • Link akan muncul di Settings → Pages

  • Contoh: https://username.github.io/studio

Dengan cara ini, kamu bisa langsung download hasil kerja kita dalam bentuk ZIP, lalu deploy ke GitHub Pages.
